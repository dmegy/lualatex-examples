\documentclass[tikz,multi]{standalone}
\usepackage{luacode}
\begin{luacode*}
-- see https://en.wikipedia.org/wiki/Cantor_function

function drawLine(P,Q)
	tex.print(
		string.format(
			"\\draw (%f,%f) -- (%f,%f);",
			P[1],P[2],Q[1],Q[2])
		)
end

function cantor_graph(P,Q, n)
	if n == 0 then
		drawLine(P,Q)
	else
		local A = {P[1]*2/3+Q[1]/3,(P[2]+Q[2])/2}
		local B = {P[1]/3+Q[1]*2/3,(P[2]+Q[2])/2}
		cantor_graph(P,A, n-1)
		drawLine(A,B)
		cantor_graph(B,Q, n-1)
	end
end

\end{luacode*}
\begin{document}

\begin{luacode*}
for k=0,7 do
	tex.print("\\begin{tikzpicture}[scale=16]")
	cantor_graph({0,0},{1,1},k)
	tex.print("\\end{tikzpicture}")
end
\end{luacode*}
\end{document}
