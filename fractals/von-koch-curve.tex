\documentclass[tikz,multi]{standalone}
\usepackage{luacode}
\begin{luacode*}

function point(O,I,J,x,y)
	-- point de coord. (x,y) dans le repère non nécessairement orthonormé (O,I,J)
	return {O[1]+x*(I[1]-O[1])+y*(J[1]-O[1]),O[2]+x*(I[2]-O[2])+y*(J[2]-O[2])}
end

function drawSegment(P,Q)
	tex.print(string.format(
		"\\draw (%f,%f) -- (%f,%f);",
		P[1],P[2],
		Q[1],Q[2]
	))
end

function koch(P,Q,R,depth)
	if depth == 0 then
		drawSegment(P,Q)
	else
		koch(P, point(P,Q,R,1/3,0), point(P,Q,R,0,1/3),depth-1)
		koch(point(P,Q,R,1/3,0), point(P,Q,R,1/3,1/3), point(P,Q,R,0,1/3), depth-1)
		koch(point(P,Q,R,1/3,1/3), point(P,Q,R,2/3,0), point(P,Q,R,2/3,1/3),depth-1)
		koch(point(P,Q,R,2/3,0), point(P,Q,R,1,0), point(P,Q,R,2/3,1/3), depth-1)
	end
end
\end{luacode*}
\begin{document}
\begin{tikzpicture}[scale=12]
\directlua{koch({0,0}, {1,0}, {.5,math.sqrt(3)/2}, 5)}
\end{tikzpicture}
\end{document}
