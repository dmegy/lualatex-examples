\begin{luacode*}
WORLD_DOWN = {0,0,-1}
WORLD_UP = {0,0,1}

linalg = {}

function linalg.norm(t)
	return math.sqrt(t[1]*t[1]+t[2]*t[2]+t[3]*t[3])
end

function linalg.normalize(t)
	local n = linalg.norm(t)
	if n == 0 then return {0,0,0} end
	return {t[1]/n, t[2]/n, t[3]/n}
end

function linalg.cross(u,v)
	return {
		u[2]*v[3]-u[3]*v[2],
		u[3]*v[1]-u[1]*v[3],
		u[1]*v[2]-u[2]*v[1]
	}
end

function linalg.dot(u,v)
	return u[1]*v[1]+u[2]*v[2]+u[3]*v[3]
end





function vec(A,B)
	return {B[1]-A[1],B[2]-A[2],B[3]-A[3]}
end

function dist(A,B)
	return linalg.norm(vec(A,B))
end

function bary(points,coefs)
	local result={0,0,0}
	for k=1,3 do
		for i=1,#points do
			result[k] =  result[k] + coefs[i]*points[i][k]
		end
	end
	return result
end

function midpoint(A,B)
	return {
		(A[1]+B[1])/2,
		(A[2]+B[2])/2,
		(A[3]+B[3])/2
	}
end



function von_koch_surface(A, B, C, depth)
	-- A, B, C,  : base, 
	
	if depth == 0 then 
		table.insert(triangles,{A, B, C})
		return
	end
	
	local H = bary({A,B,C},{1/3,1/3,1/3})
	
	local normal = linalg.normalize(
		linalg.cross(vec(A,B),vec(A,C))
	)
	local length = dist(A,B)
	
	local D = {
		H[1]+normal[1]*length/math.sqrt(6),
		H[2]+normal[2]*length/math.sqrt(6),
		H[3]+normal[3]*length/math.sqrt(6)
	}
	
	local AB = midpoint(A,B)
	local AC = midpoint(A,C)
	local BC = midpoint(B,C)
	
	
	von_koch_surface(A, AB, AC, depth-1)
	von_koch_surface(B, BC, AB, depth-1)
	von_koch_surface(C, AC, BC, depth-1)
	
	von_koch_surface(D, AC, AB,  depth-1)
	von_koch_surface(D, AB, BC,  depth-1)
	von_koch_surface(D, BC, AC,  depth-1)
	
end




function project_polygon(polygon)
	local projected_polygon = {}
	
	for _, point in ipairs(polygon) do
		table.insert(projected_polygon ,  {
			linalg.dot(camera_matrix.u,point),
			linalg.dot(camera_matrix.v,point),
			linalg.dot(camera_matrix.w,point)
		})
	end
	return projected_polygon
end

function draw_triangle(triangle)
	-- triangle is already projected
	local triangle_normal = linalg.normalize(
		linalg.cross(
			vec(triangle[2],triangle[1]),
			vec(triangle[3],triangle[1])
		)
	)
	light = math.abs(linalg.dot(triangle_normal,linalg.normalize({-1,1,3}))) -- lumière en haut à gauche
	percent = 30+math.round(light*65)
	
	texio.write_nl(light)
	options = "color=white!" .. percent .. "!black"
	tex.print(string.format("\\fill[%s]",options))
	for k=1,3 do
		tex.sprint(string.format("(%f,%f) -- ", triangle[k][1], triangle[k][2]))
	end
	tex.sprint(" cycle;")
end

triangles={}
von_koch_surface({1,0,0},{-1/2,math.sqrt(3)/2,0},{-1/2,-math.sqrt(3)/2,0}, 3)

for d=0,360,10 do
	camera_position = {math.cosd(d),math.sind(d),.5}
	camera_matrix = { w = linalg.normalize(camera_position) }
	camera_matrix.u = linalg.normalize(linalg.cross(camera_matrix.w, WORLD_DOWN))
	camera_matrix.v = linalg.normalize(linalg.cross(camera_matrix.w,camera_matrix.u))
	texio.write_nl("calculs camera")


	

	projected_triangles={}
	for k,triangle in ipairs(triangles) do
		table.insert(projected_triangles,project_polygon(triangle))
	end

	table.sort(projected_triangles, function(a,b) return a[1][3]+a[2][3]+a[3][3] < b[1][3]+b[2][3]+b[3][3] end)
	
	tex.print("\\begin{tikzpicture}[scale=12]")
	tex.print("\\fill[white] (0,0) circle (1.5);")
	for k,triangle in ipairs(projected_triangles) do
		draw_triangle(triangle)
	end
	tex.print("\\end{tikzpicture}")
	
end
