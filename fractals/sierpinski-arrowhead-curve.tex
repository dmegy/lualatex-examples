\usepackage{luacode}

\begin{luacode*}
function lerp(P,Q,t)
	return {P[1]*(1-t) + Q[1] * t, P[2]*(1-t) + Q[2] * t}
end

function drawLine(P,Q)
	tex.print(
		string.format(
			"\\draw[%s] (%f,%f) -- (%f,%f);",
			"black",
			P[1], P[2],
			Q[1], Q[2]
		)
	)
end

function sierpinski_arrowhead(P,Q,R,n)
	if n == 0 then
		drawLine(P,Q)
	else
		sierpinski_arrowhead(lerp(P,R,1/2), P, lerp(P,Q,1/2), n-1)
		sierpinski_arrowhead(lerp(P,R,1/2), lerp(Q,R,1/2), R, n-1)
		sierpinski_arrowhead(Q, lerp(R,Q,1/2), lerp(P,Q,1/2), n-1)
	end
end

\end{luacode*}
