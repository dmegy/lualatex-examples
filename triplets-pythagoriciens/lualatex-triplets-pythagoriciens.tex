\documentclass[tikz]{standalone}
\usepackage{luacode}
\begin{document}
\begin{tikzpicture}
\begin{luacode*}
local N = 1500
local M = 4000
local scale = 0.004
local POINT_RADIUS = 0.025
local squares = {}
local max = math.floor(math.sqrt(M*M + N*N))
for i = 1,max  do
  squares[i*i] = true
end

local function draw_point(x,y)
	return string.format(
        "\\fill[black] (%.3f,%.3f) circle (%.4f);",
        x,y,POINT_RADIUS
      )
end

local linesToPrint = {}

for x = 1, M do
  for y = 1, N do
    local s = x*x + y*y
    if squares[s] then
    	table.insert(linesToPrint,draw_point(x*scale,y*scale))
    end
  end
end

tex.print(table.concat(linesToPrint," "))
\end{luacode*}
\end{tikzpicture}
\end{document}
