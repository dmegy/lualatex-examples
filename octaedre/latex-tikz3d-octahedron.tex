  axedeux/.style={
    red,
    thick,
    {Circle[fill=red, length=4pt]}-{Circle[fill=red, length=4pt]}
  }
  }
  \tikzset{
  axequatre/.style={
    blue,
    thick,
    {Circle[fill=blue, length=4pt]}-{Circle[fill=blue, length=4pt]}
  }
}
  \tikzset{
  axetrois/.style={
    green,
    thick,
    {Circle[fill=green, length=4pt]}-{Circle[fill=green, length=4pt]}
  }
}
